Get-Content .env | ForEach-Object {
  if ($_ -match '^\s*([A-Za-z_][A-Za-z0-9_]*)=(.*)$') {
    $value = $matches[2] -replace '^"(.*)"$', '$1'
    Set-Item -Path "Env:$($matches[1])" -Value $value
  }
}
